class Solution {
public:
    int longestSemiRepetitiveSubstring(string s) {
        int maxSize = 1, sameCount = 0, n = s.size();
        for (int l = 0, r = 1; r < s.size(); r++) {
            if (s[r] == s[r - 1]) sameCount++;
            while (sameCount > 1) {
                if (s[l] == s[++l]) sameCount--;
            }
            maxSize = max(maxSize, r - l + 1);
        }
        return maxSize;
    }
};
