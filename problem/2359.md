class Solution {
public:
    void bfs(int startNode, vector<int>& edges, vector<int>& dist) {
        int n = edges.size();
        queue<int> q;
        q.push(startNode);

        vector<bool> visit(n);
        dist[startNode] = 0;

        while (!q.empty()) {
            int node = q.front();
            q.pop();
            if (visit[node]) continue;
            visit[node] = true;
            int neighbor = edges[node];
            if (neighbor != -1 && !visit[neighbor]) {
                dist[neighbor] = 1 + dist[node];
                q.push(neighbor);
            }
        }
    }
    int closestMeetingNode(vector<int>& edges, int node1, int node2) {
        int n = edges.size();
        vector<int> dist1(n, INT_MAX), dist2(n, INT_MAX);
        bfs(node1, edges, dist1);
        bfs(node2, edges, dist2);

        int minNode = -1, minDist = INT_MAX;
        for (int i = 0; i < n; i++) {
            if (minDist > max(dist1[i], dist2[i])) {
                minNode = i;
                minDist = max(dist1[i], dist2[i]);
            }
        }
        return minNode;
    }
};
