```c++
class Solution {
public:
    int countCompleteSubstrings(string word, int k) {
        int ans = 0;
        for (int i = 1; i <= 26; i++) {
            int l = 0;
            int m = i * k;
            vector<int> cnt(26);
            int diff = 0;
            for (int r = 0; r < word.size(); r++) {
                if (++cnt[word[r] - 'a'] == 1)
                    diff++;

                while (l < r && (cnt[word[r] - 'a'] > k ||
                                 abs(word[r] - word[r - 1]) > 2)) {
                    if (--cnt[word[l++] - 'a'] == 0)
                        diff--;
                }
                if (r - l + 1 == m) {
                    if (diff == i)
                        ans++;

                    if (--cnt[word[l++] - 'a'] == 0)
                        diff--;
                }
            }
        }
        return ans;
    }
};
```
