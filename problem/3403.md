```c++
class Solution {
public:
    string answerString(string word, int numFriends) {
        if (numFriends == 1)
            return word;
        int n = word.size();
        int m = n - numFriends + 1;
        string ans = "";
        for (int l = 0; l < n; l++) {
            ans = max(ans, word.substr(l, min(m, n - l)));
        }
        return ans;
    }
};
```
