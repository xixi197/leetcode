```c++
class Solution {
public:
    bool hasAllCodes(string s, int k) {
        int n = 1 << k;
        vector<int> nums(n, 1);
        int c = 0;
        for (int r = 0; r < s.size(); r++) {
            c = (c << 1) + s[r] - '0';
            if (r >= k - 1) {
                if (--nums[c] == 0 && --n == 0)
                    return true;
                c &= (1 << (k - 1)) - 1;
            }
        }
        return false;
    }
};
```
