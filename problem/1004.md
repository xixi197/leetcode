```c++
class Solution {
public:
    int longestOnes(vector<int>& nums, int k) {
        int maxSize = 0;
        int zeroCount = 0;
        for (int l = 0, r = 0; r < nums.size(); r++) {
            if (nums[r] == 0) zeroCount++;
            while (zeroCount > k) {
                if (nums[l++] == 0) zeroCount--;
            }
            maxSize = max(maxSize, r - l + 1);
        }
        return maxSize;
    }
};
```
