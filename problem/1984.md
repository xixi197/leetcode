```c++
class Solution {
public:
    int minimumDifference(vector<int>& nums, int k) {
        sort(nums.begin(), nums.end());
        int minDiff = INT_MAX;
        for (int r = k - 1; r < nums.size(); r++) {
            minDiff = min(minDiff, nums[r] - nums[r - k + 1]);
            if (minDiff == 0) return 0;
        }
        return minDiff;
    }
};
```
