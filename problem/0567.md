```c++
class Solution {
public:
    bool checkInclusion(string s1, string s2) {
        vector<int> count(26);
        int s = 0;
        for (auto& c : s1) {
            if (count[c - 'a']++ == 0) s++;
        }
        int n = s1.size();
        for (int l = 0, r = 0; r < s2.size(); r++) {
            if (--count[s2[r] - 'a'] == 0 && --s == 0)
                return true;
            if (r - l + 1 == n) {
                if (++count[s2[l++] - 'a'] == 1) s++;
            }
        }
        return false;
    }
};
```
