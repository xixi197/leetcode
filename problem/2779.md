```c++
class Solution {
public:
    int maximumBeauty(vector<int>& nums, int k) {
        sort(nums.begin(), nums.end());
        int maxSize = 0;
        for (int l = 0, r = 0; r < nums.size(); r++) {
            while (nums[r] - 2 * k > nums[l]) l++;
            maxSize = max(maxSize, r - l + 1);
        }
        return maxSize;
    }
};
```
