```c++
class Solution {
public:
    int longestPalindrome(vector<string>& words) {
        unordered_map<string, int> mp;
        int ret = 0;
        int two = 0;
        for (auto w : words) {
            string t = {w[1], w[0]};
            if (mp[t]) {
                ret += 4;
                mp[t]--;
                two -= (w[0] == w[1]) ? 1 : 0;
            } else {
                mp[w]++;
                two += (w[0] == w[1]) ? 1 : 0;
            }
        }
        return ret + (two > 0 ? 2 : 0);
    }
};
```
