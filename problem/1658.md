```c++
class Solution {
public:
    int minOperations(vector<int>& nums, int x) {
        int target = 0, n = nums.size();
        for (auto v : nums) {
            target += v;
        }
        target -= x;
        if (target == 0)  return n;
        if (target < 0) return -1;
        int maxCount = 0, curSum = 0;
        for (int l = 0, r = 0; r < n; r++) {
            curSum += nums[r];
            while (curSum >= target) {
                if (curSum == target)
                    maxCount = max(maxCount, r - l + 1);
                curSum -= nums[l++];
            }
        }
        return maxCount == 0 ? -1 : n - maxCount;
    }
};
```
