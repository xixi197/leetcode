class Solution {
public:
    int ctoi(char c) {
        int t = c - 'a';
        if (t >= 0 && t <= 25) return t;
        t = c - 'A';
        if (t >= 0 && t <= 25) return t;
        t = c - '0';
        if (t >= 0 && t <= 9) return t + 26;
        return -1;
    }
    bool isPalindrome(string s) {
        for (int l = 0, r = s.size() - 1; l < r; l++, r--) {
            while (l < s.size() && ctoi(s[l]) == -1)
                l++;
            while (r >= 0 && ctoi(s[r]) == -1)
                r--;
            if (l < r && ctoi(s[l]) != ctoi(s[r]))
                return false;
        }
        return true;
    }
};
