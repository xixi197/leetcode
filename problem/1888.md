```c++
class Solution {
public:
    int minFlips(string s) {
        int c = 0;
        int n = s.size();
        for (int i = 0; i < n; i++) {
            c += (i % 2 == s[i] - '0') ? 1 : 0;
        }
        int minc = min(c, n - c);
        if (c == 0 || c == n || n % 2 == 0)
            return minc;
        for (int i = 0; i < n; i++) {
            c += (i % 2 == s[i] - '0') ? -1 : 1;
            minc = min(minc, min(c, n - c));
        }
        return minc;
    }
};
```
