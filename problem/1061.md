class Solution {
public:
    string smallestEquivalentString(string s1, string s2, string baseStr) {
        vector<int> v(26);
        for (int i = 0; i < 26; i++) {
            v[i] = i;
        }
        for (int i = 0; i < s1.size(); i++) {
            int c1 = s1[i] - 'a';
            int c2 = s2[i] - 'a';
            int l1 = c1;
            int l2 = c2;
            while (l1 != v[l1]) {
                l1 = v[l1];
            }
            while (l2 != v[l2]) {
                l2 = v[l2];
            }
            if (l1 < l2) {
                v[l2] = l1;
            } else {
                v[l1] = l2;
            }
        }

        for (int i = 0; i < baseStr.size(); i++) {
            int j = baseStr[i] - 'a';
            while (j != v[j]) {
                j = v[j];
            }
            baseStr[i] = 'a' + j;
        }
        return baseStr;
    }
};
